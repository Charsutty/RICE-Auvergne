# RICE: Auvergne — SCRIPTED TRIGGERS

AUV_is_in_flavor_scope_trigger = {
	OR = {
		has_title = title:d_auvergne
		has_title = title:c_auvergne
		has_title = title:c_clermont_sur_allier
		has_title = title:c_thiers
		has_title = title:c_velay
		has_title = title:c_gevaudan
	}
}

AUV_controls_clermont_scope_trigger = {
	OR = {
		has_title = title:c_auvergne
		has_title = title:c_clermont_sur_allier
	}
}

AUV_controls_comital_auvergne_scope_trigger = {
	OR = {
		has_title = title:d_auvergne
		has_title = title:c_auvergne
		has_title = title:c_clermont_sur_allier
	}
}

AUV_controls_auvergnat_patronage_scope_trigger = {
	OR = {
		has_title = title:d_auvergne
		has_title = title:c_auvergne
		has_title = title:c_clermont_sur_allier
		has_title = title:c_thiers
	}
}

AUV_controls_upper_allier_scope_trigger = {
	OR = {
		has_title = title:c_velay
		has_title = title:c_gevaudan
	}
}

AUV_can_negotiate_clermont_authority_trigger = {
	AUV_controls_clermont_scope_trigger = yes
	NOT = { has_character_flag = AUV_clermont_authority_recently_settled }
}

AUV_can_address_comital_succession_trigger = {
	AUV_controls_comital_auvergne_scope_trigger = yes
	current_date >= 1100.1.1
	exists = dynasty
	dynasty = {
		NOR = {
			has_dynasty_modifier = AUV_comital_house_compact_modifier
			has_dynasty_modifier = AUV_divided_house_foundation_modifier
		}
	}
	any_close_family_member = {
		is_alive = yes
		is_adult = yes
	}
}

AUV_can_secure_upper_allier_routes_trigger = {
	AUV_controls_upper_allier_scope_trigger = yes
	OR = {
		AND = {
			has_title = title:c_velay
			title:c_velay = {
				NOR = {
					has_county_modifier = AUV_upper_allier_routes_modifier
					has_county_modifier = AUV_upper_allier_tolls_modifier
					has_county_modifier = AUV_upper_allier_compact_modifier
				}
			}
		}
		AND = {
			has_title = title:c_gevaudan
			title:c_gevaudan = {
				NOR = {
					has_county_modifier = AUV_upper_allier_routes_modifier
					has_county_modifier = AUV_upper_allier_tolls_modifier
					has_county_modifier = AUV_upper_allier_compact_modifier
				}
			}
		}
	}
}
