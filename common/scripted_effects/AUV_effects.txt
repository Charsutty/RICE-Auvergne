AUV_clear_clermont_authority_character_modifiers_effect = {
	if = {
		limit = { has_character_modifier = AUV_clermont_ecclesiastical_compact_modifier }
		remove_character_modifier = AUV_clermont_ecclesiastical_compact_modifier
	}
	if = {
		limit = { has_character_modifier = AUV_clermont_comital_ascendancy_modifier }
		remove_character_modifier = AUV_clermont_comital_ascendancy_modifier
	}
	if = {
		limit = { has_character_modifier = AUV_clermont_balanced_authority_modifier }
		remove_character_modifier = AUV_clermont_balanced_authority_modifier
	}
}

AUV_clear_clermont_authority_county_modifiers_effect = {
	if = {
		limit = { has_county_modifier = AUV_clermont_privileges_county_modifier }
		remove_county_modifier = AUV_clermont_privileges_county_modifier
	}
	if = {
		limit = { has_county_modifier = AUV_clermont_jurisdiction_county_modifier }
		remove_county_modifier = AUV_clermont_jurisdiction_county_modifier
	}
	if = {
		limit = { has_county_modifier = AUV_clermont_compromise_county_modifier }
		remove_county_modifier = AUV_clermont_compromise_county_modifier
	}
}

AUV_clermont_confirm_privileges_effect = {
	AUV_clear_clermont_authority_character_modifiers_effect = yes
	add_character_modifier = {
		modifier = AUV_clermont_ecclesiastical_compact_modifier
		years = 10
	}
	add_piety = minor_piety_gain

	if = {
		limit = { has_title = title:c_auvergne }
		title:c_auvergne = {
			AUV_clear_clermont_authority_county_modifiers_effect = yes
			add_county_modifier = {
				modifier = AUV_clermont_privileges_county_modifier
				years = 10
			}
		}
	}
	if = {
		limit = { has_title = title:c_clermont_sur_allier }
		title:c_clermont_sur_allier = {
			AUV_clear_clermont_authority_county_modifiers_effect = yes
			add_county_modifier = {
				modifier = AUV_clermont_privileges_county_modifier
				years = 10
			}
		}
	}
}

AUV_clear_upper_allier_modifiers_effect = {
	if = {
		limit = { has_county_modifier = AUV_upper_allier_routes_modifier }
		remove_county_modifier = AUV_upper_allier_routes_modifier
	}
	if = {
		limit = { has_county_modifier = AUV_upper_allier_tolls_modifier }
		remove_county_modifier = AUV_upper_allier_tolls_modifier
	}
	if = {
		limit = { has_county_modifier = AUV_upper_allier_compact_modifier }
		remove_county_modifier = AUV_upper_allier_compact_modifier
	}
}
